'border-red-200 bg-red-50/30' : ''}`}><div className="flex items-center justify-between mb-2"><div className="flex items-center gap-2 text-sm"><button onClick={() => setAdminViewUser({ id: r.authorId, name: r.author })} className="font-medium text-teal-600 hover:underline">{r.author}</button><span className="text-gray-400">{r.date}</span>{r.flagged && <span className="bg-red-100 text-red-600 px-2 py-0.5 rounded text-xs">Flagged</span>}</div><div className="flex items-center gap-2">{r.flagged && <button onClick={() => unflagReply(adminViewThread.id, r.id)} className="text-emerald-600 hover:bg-emerald-50 p-1 rounded"><CheckCircle size={16}/></button>}<button onClick={() => deleteReply(adminViewThread.id, r.id)} className="text-red-500 hover:bg-red-50 p-1 rounded"><Trash2 size={16}/></button></div></div><p className="text-gray-700">{r.text}</p><div className="flex items-center gap-2 mt-2 text-sm text-gray-500"><ThumbsUp size={12}/>{r.upvotes}<ThumbsDown size={12} className="ml-2"/>{r.downvotes}</div></div>)}</div>
          </div>
        </div>
      );
    }

    return (
    <div className="space-y-6">
      {(flaggedPosts.length > 0 || flaggedReplies.length > 0) && <div className="bg-white rounded-2xl shadow-lg overflow-hidden"><div className="px-6 py-4 border-b bg-red-50 flex items-center gap-3"><Flag className="text-red-500"/><h2 className="text-xl font-bold text-red-800">Flagged ({flaggedPosts.length + flaggedReplies.length})</h2></div><div className="p-4 space-y-3">{flaggedPosts.map(p => <div key={p.id} className="p-4 border border-red-200 bg-red-50/50 rounded-xl flex justify-between items-start"><div><div className="flex items-center gap-2 mb-1"><span className="bg-red-100 text-red-600 px-2 py-0.5 rounded text-xs">{p.status === 'pending' ? 'Pending' : 'Flagged'}</span><span className="text-gray-500 text-sm">{p.author}</span></div><h3 className="font-semibold text-gray-800">{p.title}</h3><p className="text-gray-600 text-sm">{p.content}</p></div><div className="flex gap-2"><button onClick={() => setAdminViewThread(p)} className="text-teal-600 hover:bg-teal-50 p-2 rounded-lg"><Eye size={18}/></button><button onClick={() => approvePost(p.id)} className="text-emerald-600 hover:bg-emerald-50 p-2 rounded-lg"><CheckCircle size={18}/></button><button onClick={() => deletePost(p.id)} className="text-red-500 hover:bg-red-50 p-2 rounded-lg"><Trash2 size={18}/></button></div></div>)}{flaggedReplies.map((r,i) => <div key={i} className="p-4 border border-orange-200 bg-orange-50/50 rounded-xl flex justify-between items-start"><div><div className="flex items-center gap-2 mb-1"><span className="bg-orange-100 text-orange-600 px-2 py-0.5 rounded text-xs">Flagged Reply</span><span className="text-gray-500 text-sm">on "{r.postTitle}"</span></div><p className="text-gray-600">"{r.text}" — {r.author}</p></div><div className="flex gap-2"><button onClick={() => setAdminViewThread(forumPosts.find(p => p.id === r.postId))} className="text-teal-600 hover:bg-teal-50 p-2 rounded-lg"><Eye size={18}/></button><button onClick={() => unflagReply(r.postId, r.id)} className="text-emerald-600 hover:bg-emerald-50 p-2 rounded-lg"><CheckCircle size={18}/></button><button onClick={() => deleteReply(r.postId, r.id)} className="text-red-500 hover:bg-red-50 p-2 rounded-lg"><Trash2 size={18}/></button></div></div>)}</div></div>}
      <div className="bg-white rounded-2xl shadow-lg overflow-hidden"><div className="px-6 py-4 border-b flex items-center gap-3"><MessageSquare className="text-teal-600"/><h2 className="text-xl font-bold text-gray-800">All Posts</h2></div><div className="divide-y">{forumPosts.map(p => <div key={p.id} className="p-4 hover:bg-gray-50 flex justify-between items-center"><div className="flex-1"><div className="flex items-center gap-2 mb-1"><span className={`px-2 py-0.5 rounded text-xs ${p.status === 'approved' ? 'bg-emerald-100 text-emerald-700' : 'bg-orange-100 text-orange-700'}`}>{p.status}</span><button onClick={() => setAdminViewUser({ id: p.authorId, name: p.author })} className="text-sm text-teal-600 hover:underline">{p.author}</button><span className="text-gray-400 text-sm">{p.date}</span></div><h3 className="font-semibold text-gray-800">{p.title}</h3><p className="text-gray-500 text-sm">{p.replies.length} replies • {p.upvotes}↑ {p.downvotes}↓</p></div><div className="flex gap-2"><button onClick={() => setAdminViewThread(p)} className="text-teal-600 hover:bg-teal-50 p-2 rounded-lg"><Eye size={18}/></button>{p.status !== 'approved' && <button onClick={() => approvePost(p.id)} className="text-emerald-600 hover:bg-emerald-50 p-2 rounded-lg"><CheckCircle size={18}/></button>}<button onClick={() => deletePost(p.id)} className="text-red-500 hover:bg-red-50 p-2 rounded-lg"><Trash2 size={18}/></button></div></div>)}</div></div>
      <div className="bg-white rounded-2xl shadow-lg overflow-hidden"><div className="px-6 py-4 border-b flex items-center gap-3"><Users className="text-cyan-600"/><h2 className="text-xl font-bold text-gray-800">Forum Users</h2></div><div className="p-4 flex flex-wrap gap-2">{allAuthors.map((a,i) => <button key={i} onClick={() => setAdminViewUser(a)} className="px-3 py-2 bg-gray-100 hover:bg-teal-100 rounded-lg text-sm flex items-center gap-2"><User size={14}/>{a.name}<ChevronRight size={14}/></button>)}</div></div>
    </div>
  )};

  const AdminMailing = () => (
    <div className="bg-white rounded-2xl shadow-lg overflow-hidden"><div className="px-6 py-4 border-b flex items-center gap-3"><Mail className="text-emerald-600"/><h2 className="text-xl font-bold text-gray-800">Mailing List ({mailingList.length})</h2></div><div className="p-6 flex flex-wrap gap-3">{mailingList.map((e,i) => <div key={i} className="bg-emerald-50 text-emerald-700 px-4 py-2 rounded-full flex items-center gap-2 text-sm">{e}<button onClick={() => removeMail(e)} className="hover:bg-emerald-200 rounded-full p-1"><X size={14}/></button></div>)}</div></div>
  );

  const AdminView = () => (
    <div className="min-h-screen bg-gray-100">
      <div className="bg-gradient-to-r from-slate-800 to-slate-700 text-white px-6 py-4"><div className="max-w-7xl mx-auto flex items-center justify-between"><div className="flex items-center gap-3"><Droplets size={28} className="text-teal-400"/><h1 className="text-xl font-bold">Admin</h1></div><div className="flex gap-3 text-sm"><span className="bg-teal-500/20 text-teal-300 px-3 py-1 rounded-full">{users.length} Volunteers</span><span className="bg-orange-500/20 text-orange-300 px-3 py-1 rounded-full">{allUserDocs.filter(d => d.status === 'pending').length} Pending</span></div></div></div>
      <div className="max-w-7xl mx-auto px-6 py-6"><div className="flex gap-2 mb-6 bg-white rounded-xl p-1.5 shadow-sm overflow-x-auto">{adminTabs.map(t => <button key={t.id} onClick={() => { setAdminTab(t.id); setAdminViewThread(null); setAdminViewUser(null); }} className={`flex items-center gap-2 px-4 py-2 rounded-lg text-sm font-medium whitespace-nowrap ${adminTab === t.id ? 'bg-teal-500 text-white' : 'text-gray-600 hover:bg-gray-100'}`}><t.icon size={18}/>{t.label}</button>)}</div>{adminTab === 'volunteers' && <AdminVolunteers/>}{adminTab === 'events' && <AdminEvents/>}{adminTab === 'documents' && <AdminDocuments/>}{adminTab === 'forum' && <AdminForum/>}{adminTab === 'mailing' && <AdminMailing/>}</div>
      {showEventModal && <EventFormModal/>}
    </div>
  );

  return (
    <div>
      <div className="fixed top-4 right-4 z-50 bg-white rounded-full shadow-lg p-1 flex"><button onClick={() => setView('user')} className={`px-4 py-2 rounded-full text-sm font-medium ${view === 'user' ? 'bg-teal-500 text-white' : 'text-gray-600 hover:bg-gray-100'}`}>User</button><button onClick={() => setView('admin')} className={`px-4 py-2 rounded-full text-sm font-medium ${view === 'admin' ? 'bg-slate-700 text-white' : 'text-gray-600 hover:bg-gray-100'}`}>Admin</button></div>
      {view === 'user' ? <UserView/> : <AdminView/>}
    </div>
  );
}
